// Supabase Edge Function: analyze-case // Runs Hive's AI-Generated & Deepfake Detection on an uploaded case file, // saves the result to the database, and returns a summary to the portal. // // SECURITY: the Hive key lives ONLY here, as a Supabase secret (HIVE_KEY). // It is never sent to the browser. This function can only be called by a // logged-in analyst (Supabase verifies their login token automatically). import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; // ===== Hive config ===== // V3 endpoint for the AI-Generated & Deepfake Content Detection model: const HIVE_ENDPOINT = "https://api.thehive.ai/api/v3/hive/ai-generated-and-deepfake-content-detection"; // 2) Your Hive Secret Key, stored as a Supabase secret (you set this in the dashboard): const HIVE_KEY = Deno.env.get("HIVE_KEY") ?? ""; // 3) Hive's V3 APIs authenticate with Bearer. (V2 legacy APIs use "Token" instead.) const HIVE_AUTH_HEADER = `Bearer ${HIVE_KEY}`; // ===== Resemble config (audio) ===== const RESEMBLE_ENDPOINT = "https://app.resemble.ai/api/v2/detect"; const RESEMBLE_KEY = Deno.env.get("RESEMBLE_API_KEY") ?? ""; const AUDIO_EXTS = ["mp3", "wav", "m4a", "ogg", "flac", "aac", "amr", "3gp", "3gpp"]; Deno.serve(async (req) => { // Browser preflight if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); try { const { case_ref, file_path } = await req.json(); if (!file_path) return json({ error: "file_path is required" }, 400); // Server-side admin client: reads the private file + writes results. // SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY are injected automatically. const admin = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, ); // 1) Download the file bytes from Supabase (server-side, service role) const { data: blob, error: dlErr } = await admin .storage.from("case-files").download(file_path); if (dlErr || !blob) { const available = await listAllFiles(admin); return json({ error: "could not download file: " + (dlErr?.message ?? "unknown"), you_sent: file_path, available_files: available, }, 200); } // 2) Route by file type: audio -> Resemble, everything else -> Hive. const filename = (file_path.split("/").pop() || "upload"); const ext = (filename.split(".").pop() || "").toLowerCase(); const engine = AUDIO_EXTS.includes(ext) ? "resemble" : "hive"; const r: any = engine === "resemble" ? await runResemble(blob, filename) : await runHive(blob, filename); if (r.error) { return json({ debug: engine + "_error", status: r.status, response: r.body }, 200); } // 3) Save the tool result await admin.from("tool_results").insert({ case_ref: case_ref ?? null, tool_name: r.tool_name, tool_version: r.tool_version, score: r.manipulation_score, verdict: r.verdict, raw_output: r.raw, }); // 4) Return a clean summary (engine tells the portal which model ran) return json({ ok: true, engine, verdict: r.verdict, manipulation_score: r.manipulation_score, ai_generated: r.ai_generated, deepfake: r.deepfake, }); } catch (e) { return json({ error: String(e) }, 500); } }); // Runs Hive (image / video) on the file bytes. async function runHive(blob: Blob, filename: string) { const form = new FormData(); form.append("media", blob, filename); const res = await fetch(HIVE_ENDPOINT, { method: "POST", headers: { "Authorization": HIVE_AUTH_HEADER }, // boundary set automatically body: form, }); const text = await res.text(); let body: any = null; try { body = JSON.parse(text); } catch (_) { body = text; } if (!res.ok) return { error: true, status: res.status, body }; const { aiScore, deepfakeScore } = parseHive(body); const manipulation = Math.round(Math.max(aiScore, deepfakeScore) * 100); const verdict = (deepfakeScore >= 0.9 || aiScore >= 0.9) ? "MANIPULATED" : (deepfakeScore >= 0.5 || aiScore >= 0.5) ? "INCONCLUSIVE" : "AUTHENTIC"; return { tool_name: "Hive AI-Generated & Deepfake Detection", tool_version: "v3", verdict, manipulation_score: manipulation, ai_generated: aiScore, deepfake: deepfakeScore, raw: body, }; } // Runs Resemble Detect (audio). "Prefer: wait" makes it block until done, // so we get the result in one call with no polling. async function runResemble(blob: Blob, filename: string) { const form = new FormData(); form.append("file", blob, filename); const res = await fetch(RESEMBLE_ENDPOINT, { method: "POST", headers: { "Authorization": `Bearer ${RESEMBLE_KEY}`, "Prefer": "wait" }, body: form, }); const text = await res.text(); let body: any = null; try { body = JSON.parse(text); } catch (_) { body = text; } if (!res.ok) return { error: true, status: res.status, body }; const { fakeScore: agg, label } = parseResemble(body); const lab = (label || "").toLowerCase(); const isFake = /fake|synthetic|deepfake|spoof|generated/.test(lab); const isReal = /real|authentic|human|genuine|bona/.test(lab); // Resemble's aggregated_score is the fake probability directly (0 = real, 1 = fake). const fakeness = agg; const manipulation = Math.round(fakeness * 100); const verdict = isFake ? "MANIPULATED" : isReal ? "AUTHENTIC" : (manipulation >= 90 ? "MANIPULATED" : manipulation >= 50 ? "INCONCLUSIVE" : "AUTHENTIC"); return { tool_name: "Resemble Detect (audio)", tool_version: "v2", verdict, manipulation_score: manipulation, ai_generated: fakeness, deepfake: 0, raw: body, }; } // Finds Resemble's audio verdict label + aggregated_score anywhere in the response. function parseResemble(payload: any): { fakeScore: number; label: string } { let label = "", agg = NaN; function walk(node: any) { if (!node || typeof node !== "object") return; if (node.label !== undefined && node.aggregated_score !== undefined) { label = String(node.label); const v = parseFloat(node.aggregated_score); if (!isNaN(v)) agg = v; } for (const k of Object.keys(node)) walk(node[k]); } walk(payload); const f = isNaN(agg) ? 0 : (agg > 1 ? agg / 100 : agg); return { fakeScore: f, label }; } // Lists every file in the case-files bucket (root + one level of folders), // so we can see the exact paths that exist. async function listAllFiles(admin: any): Promise { const paths: string[] = []; const { data: top } = await admin.storage.from("case-files").list("", { limit: 100 }); for (const item of (top ?? [])) { const isFolder = item?.id === null || !item?.metadata; if (isFolder) { const { data: inner } = await admin.storage.from("case-files").list(item.name, { limit: 100 }); for (const f of (inner ?? [])) { if (f?.id !== null) paths.push(item.name + "/" + f.name); } } else { paths.push(item.name); } } return paths; } // Finds ai_generated / deepfake scores anywhere in Hive's response, // so we're robust to the exact envelope shape. function parseHive(payload: any): { aiScore: number; deepfakeScore: number } { let ai = 0, df = 0; function walk(node: any) { if (!node || typeof node !== "object") return; if (Array.isArray(node.classes)) { for (const c of node.classes) { const s = c?.value ?? c?.score ?? 0; if (c?.class === "ai_generated") ai = Math.max(ai, s); if (c?.class === "deepfake") df = Math.max(df, s); } } for (const k of Object.keys(node)) walk(node[k]); } walk(payload); return { aiScore: ai, deepfakeScore: df }; } function json(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json" }, }); }
